require('dotenv').config();
const express = require('express');
const axios = require('axios');
const app = express();

const CLIENT_ID = process.env.SPOTIFY_CLIENT_ID;
const CLIENT_SECRET = process.env.SPOTIFY_CLIENT_SECRET;
// we have to use 127.0.0.1 in accordance with Spotify's redirect URI requirements
const REDIRECT_URI = 'http://127.0.0.1:3000/callback';
// stores the user prompts in memory.
const prompts = [];
const playlists = [];

function fetchUserPlaylists(accessToken) {
    return axios.get('https://api.spotify.com/v1/me/playlists', {
        headers: {
            'Authorization': 'Bearer ' + accessToken
        }
    });
}

function fetchPlaylistTracks(accessToken, playlistId) {
    return axios.get(`https://api.spotify.com/v1/playlists/${playlistId}/tracks`, {
        headers: {
            'Authorization': 'Bearer ' + accessToken
        }
    });
}

app.use(express.json());

app.use((req, res, next) => {
    res.setHeader('Access-Control-Allow-Origin', 'http://127.0.0.1:5500');
    res.setHeader('Access-Control-Allow-Methods', 'GET,POST,OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

    if (req.method === 'OPTIONS') {
        return res.sendStatus(204);
    }

    next();
});

// 1. Redirect user to Spotify
app.get('/login', (req, res) => {
    const scope = 'user-read-private user-read-email user-library-read';
    const authUrl = `https://accounts.spotify.com/authorize?response_type=code&client_id=${CLIENT_ID}&scope=${encodeURIComponent(scope)}&redirect_uri=${encodeURIComponent(REDIRECT_URI)}`;
    res.redirect(authUrl);
});

// 2. Handle the callback from Spotify
app.get('/callback', async (req, res) => {
    const code = req.query.code || null;

    try {
        const response = await axios({
            method: 'post',
            url: 'https://accounts.spotify.com/api/token',
            data: new URLSearchParams({
                code: code,
                redirect_uri: REDIRECT_URI,
                grant_type: 'authorization_code',
            }).toString(),
            headers: {
                'content-type': 'application/x-www-form-urlencoded',
                'Authorization': 'Basic ' + (Buffer.from(CLIENT_ID + ':' + CLIENT_SECRET).toString('base64'))
            },
        });

        const { access_token } = response.data;

        // 3. The Magic: Send a script back that notifies the opener and closes itself
        res.send(`
            <script>
                // Use a try-catch block to guarantee window.close() runs even if postMessage fails
                try {
                    window.opener.postMessage({
                        type: 'SPOTIFY_AUTH_SUCCESS',
                        accessToken: '${access_token}'
                    }, 'http://127.0.0.1:5500'); // <-- Removed /callback
                    window.close();
                } catch (e) {
                    console.error("Message failed:", e);
                    window.close(); // Force close even on error
                }
            </script>
        `);
    } catch (error) {
        res.send('Authentication Error');
    }
});

// listen to prompt from the client side
app.post('/prompt', (req, res) => {
    const promptText = req.body.prompt || req.body.userPrompt;

    if (!promptText || !promptText.trim()) {
        return res.status(400).json({ message: 'Prompt is required' });
    }

    const storedPrompt = {
        text: promptText.trim(),
        createdAt: new Date().toISOString(),
    };

    prompts.push(storedPrompt);
    console.log('Received prompt:', storedPrompt.text);

    res.json({ message: 'Prompt received successfully', prompt: storedPrompt });
    
});

app.get('/playlist_descriptions', (req, res) => {
    // WIP, the server will ask for descriptions from the agent and then send them to the client to be displayed on the playlist_descriptions page
     // For now, we will just send a dummy response with the number of song lists to be displayed on the playlist_descriptions page
     const songLists = req.query.songLists || 2; // default to 2 if not provided
     res.json({ songLists: songLists });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
